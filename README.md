DATABASE SETUP 
1. Create New Database
CREATE DATABASE mandarin_alarm;
USE mandarin_alarm;

2. Create the table
CREATE TABLE HSK (
ID_HSK INT PRIMARY KEY,
HSK_LEVEL VARCHAR(20) NOT NULL
);

CREATE TABLE VOCAB (
ID_VOCAB VARCHAR(40) PRIMARY KEY,
ID_HSK INT,
Hanzi VARCHAR(30) NOT NULL,
Pinyin VARCHAR(20) NOT NULL,
Meaning VARCHAR(255) NOT NULL
);

CREATE TABLE QUIZ_LOG (
ID_LOG INT PRIMARY KEY AUTO_INCREMENT, 
CURRENT_HSK INT NOT NULL,
TOTAL_QUESTION INT NOT NULL,  
CORRECT_QUESTION INT NOT NULL,
PERCENTAGE DECIMAL(5,2) GENERATED ALWAYS AS ((CORRECT_QUESTION / TOTAL_QUESTION) * 100)
);
